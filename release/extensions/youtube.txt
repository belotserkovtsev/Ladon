# ladon extension: YouTube
#
# Подключается через config.yaml:
#   allow_extensions: [youtube]
#
# Зачем статикой, а не «движок разберётся органически»: этот класс движку не
# по зубам by design. Прямой путь к YouTube не рвётся с внятной ошибкой, а
# молча уходит в таймаут, и при этом Google непрерывно ротирует адреса.
# Движковый набор наполняется реконсайлом раз в 30 секунд, а клиент получает
# свежий адрес и уходит по нему напрямую раньше — эту гонку движок выиграть
# не может. Домены отсюда dnsmasq кладёт в ipset во время резолва, до первого
# пакета клиента.

# Основное
youtube.com
youtu.be
youtube-nocookie.com

# Видеопоток. Отдаётся с хостов вида rrN---sn-XXXXXXX.googlevideo.com, которые
# генерируются на каждую сессию — перечислить их нельзя, работает только
# корневой домен.
googlevideo.com

# Превью, аватарки, статика плеера
ytimg.com
ggpht.com

# API плеера и проверка подлинности клиента. Без них страница открывается,
# но видео не стартует.
youtubei.googleapis.com
jnn-pa.googleapis.com
